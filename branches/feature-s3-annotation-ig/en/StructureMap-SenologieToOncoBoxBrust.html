// title: Senologie FHIR Bundle to OncoBox Brust 2.0 (Master Orchestrator)
// targetSpec: OncoBox Brust 2.0 — Auditjahr 2026, Spezifikation v2.1.1
// XSD-Quelle: https://xml-oncobox.de/de/Home/FuerHersteller (Brust2/26_1_oncobox-brust xsd-schema dkgzert (251110).xsd)
// ersetzt Vorgaengerversion 1.0 / N1.1.1

// status: draft

uses "http://hl7.org/fhir/StructureDefinition/Bundle" alias Bundle as source
uses "http://hl7.org/fhir/StructureDefinition/Patient" alias Patient as source
uses "http://hl7.org/fhir/StructureDefinition/Encounter" alias Encounter as source
uses "http://hl7.org/fhir/StructureDefinition/Condition" alias Condition as source
uses "http://hl7.org/fhir/StructureDefinition/Procedure" alias Procedure as source

// ============================================================================
// Master Orchestrator: Bundle -> OncoBox Brust 2.0 Meldung
// Struktur in 2.0:
// OncoBoxBrust
// ├─ ImportInfo (DatumXML, NameTudokusys, VersionTudokusys)
// └─ Patienten
// └─ Patient[]
// ├─ Stammdaten (OncoBoxID, Geburtsjahr, Geschlecht, …)
// └─ Fall[]
// ├─ Fallinformation (FallID, Falldokumentation)
// ├─ Anamnese (BRCA Keimbahn, RaucherStatus)
// ├─ Diagnose (ICD, Seitenlokalisation, cTNM, pTNM)
// ├─ Tumorkonferenz[]
// ├─ Histologie[] (Tumortyp, Grading, ER/PR/HER2/Ki67/PD-L1)
// ├─ Operation[]
// ├─ Strahlentherapie[]
// ├─ Systemtherapie[]
// ├─ Fernmetastase[]
// ├─ Studie[]
// ├─ Psychosozial
// └─ Follow_Up[]
// Codes-Conventions:
// Geschlecht: FHIR female=W, male=M, other=D, unknown=X
// Versicherungsform: gesetzlich=G, privat=P
// Seitenlokalisation: L|R|B|S (links|rechts|beidseits|ohne Seitenang.)
// Diagnosesicherheit: 0–7 (oBDS-konform); Histo=H/X (post-OP)
// ============================================================================
group SenologieToOncoBoxBrust(source src : Bundle, target tgt : OncoBoxBrust) {
  // === ImportInfo (Meldungs-Metadaten) ===
  src -> tgt.ImportInfo as ii then {
    src -> ii.DatumXML = '2026-05-13' "SetDatumXML";
    src -> ii.NameTudokusys = 'Senologie-KDS FHIR IG' "SetNameTudokusys";
    src -> ii.VersionTudokusys = '0.9.0' "SetVersionTudokusys";
  } "MapImportInfo";
  // === Patienten-Container ===
  src -> tgt.Patienten as ps then {
    src.entry as patEntry where (resource.is(Patient)) then {
      patEntry.resource as pat -> ps.Patient as p then {
        // --- Stammdaten ---
        pat -> p.Stammdaten as sd then {
          pat.id as pid -> sd.OncoBoxID = pid "SetOncoBoxID";
          pat.birthDate as bd -> sd.Geburtsjahr = bd "SetGeburtsjahr";
          pat.gender as g then {
            g where ($this = 'female') -> sd.Geschlecht = 'W' "SetGeschlechtW";
            g where ($this = 'male') -> sd.Geschlecht = 'M' "SetGeschlechtM";
            g where ($this = 'other') -> sd.Geschlecht = 'D' "SetGeschlechtD";
            g where ($this = 'unknown') -> sd.Geschlecht = 'X' "SetGeschlechtX";
          } "MapGender";
          pat.deceased as dec where ($this.is(dateTime)) -> sd.Sterbedatum = dec "SetSterbedatum";
        } "MapStammdaten";
        // --- Fall: pro Condition (Brust-Diagnose) ein Fall-Block ---
        src.entry as condEntry where (resource.is(Condition) and resource.meta.profile.exists($this.contains('senologie-diagnose-maligne'))) then {
          condEntry.resource as cond -> p.Fall as f then {
            // -- Fallinformation --
            cond -> f.Fallinformation as fi then {
              cond.id as cid -> fi.FallID = cid "SetFallID";
              cond -> fi.Falldokumentation = 'A' "SetFalldokA";
            } "MapFallinfo";
            // -- Anamnese (BRCA Keimbahn, Komorbiditaeten, Menopausenstatus, RaucherStatus) --
            cond -> f.Anamnese as an then {
              // Krebsvorerkrankungen aus Condition.evidence oder Anamnese-Obs
              src.entry as vorEntry where (resource.is(Condition) and resource.category.coding.exists(code = 'problem-list-item')) then {
                vorEntry.resource as vorCond -> an.Krebsvorerkrankungen = 'J' "SetKrebsvor";
              } "EntryVorerkrankungen";
              // RaucherStatus aus ISiK Raucher-Observation (LOINC 72166-2)
              // Werte sind LOINC LA-Codes (Answer List LL2828-3)
              src.entry as rauchEntry where (resource.is(Observation) and resource.code.coding.exists(code = '72166-2')) then {
                rauchEntry.resource as rauchObs then {
                  rauchObs.value as val then {
                    val.coding as c then {
                      c.code as cd where ($this = 'LA18976-3') -> an.RaucherStatus = 'R' "SetRauchR";
                      c.code as cd where ($this = 'LA18977-1') -> an.RaucherStatus = 'G' "SetRauchG";
                      c.code as cd where ($this = 'LA15920-4') -> an.RaucherStatus = 'Ex' "SetRauchEx";
                      c.code as cd where ($this = 'LA18978-9') -> an.RaucherStatus = 'N' "SetRauchN";
                    } "MapRauchCoding";
                  } "MapRauchVal";
                } "MapRauchObs";
              } "EntryRaucher";
              // Menopausenstatus: abgeleitet aus Gynaekologische Anamnese
              // Wenn component 42802-9 (Age at menopause) vorhanden → postmenopausal (3)
              src.entry as gynEntry where (resource.is(Observation) and resource.code.coding.exists(code = '89221-6')) then {
                gynEntry.resource as gynObs then {
                  gynObs.component as comp where (code.coding.exists(code = '42802-9')) -> an.Praetherapeutischer_Menopausenstatus = '3' "SetMenoPost";
                } "MapGynObs";
              } "EntryGyn";
              // BRCA1/2/PALB2 Keimbahn-Observations (MII Onko Genetische Variante, LOINC 69548-6)
              // Gen ist als component (48018-6, valueCodeableConcept HGNC) codiert.
              // value (Observation.valueCodeableConcept) ist das Gesamtergebnis (positive/negative).
              src.entry as brca1Entry where (resource.is(Observation) and resource.code.coding.exists(code = '69548-6') and resource.component.value.coding.exists(code = 'HGNC:1100')) then {
                brca1Entry.resource as brca1Obs then {
                  brca1Obs.value as val then {
                    val.coding as c then {
                      c.code as cd where ($this = 'LA9633-4') -> an.BRCA1_Keimbahn = 'P' "SetBRCA1Pos";
                      c.code as cd where ($this = 'LA9634-2') -> an.BRCA1_Keimbahn = 'N' "SetBRCA1Neg";
                    } "MapBRCA1Coding";
                  } "MapBRCA1Val";
                } "MapBRCA1Obs";
              } "EntryBRCA1";
              src.entry as brca2Entry where (resource.is(Observation) and resource.code.coding.exists(code = '69548-6') and resource.component.value.coding.exists(code = 'HGNC:1101')) then {
                brca2Entry.resource as brca2Obs then {
                  brca2Obs.value as val then {
                    val.coding as c then {
                      c.code as cd where ($this = 'LA9633-4') -> an.BRCA2_Keimbahn = 'P' "SetBRCA2Pos";
                      c.code as cd where ($this = 'LA9634-2') -> an.BRCA2_Keimbahn = 'N' "SetBRCA2Neg";
                    } "MapBRCA2Coding";
                  } "MapBRCA2Val";
                } "MapBRCA2Obs";
              } "EntryBRCA2";
              // Krebsvorerkrankungen: wenn weitere Conditions im Bundle (mit anderen ICD-10-Codes als der Hauptdiagnose)
              src.entry as komorbEntry where (resource.is(Condition) and resource.meta.profile.exists($this.contains('komorbiditaet'))) then {
                komorbEntry.resource as kc then {
                  kc.code as kcc -> an.KomorbiditaetenVorDiagnose as kvd then {
                    kcc.coding as c where (system = 'http://fhir.de/CodeSystem/bfarm/icd-10-gm') then {
                      c.code as cd -> kvd.Wert = cd "SetKomorbWert";
                    } "MapKomorbCoding";
                  } "MapKomorbCode";
                } "MapKomorbCond";
              } "EntryKomorb";
            } "MapAnamnese";
            // -- Diagnose --
            cond -> f.Diagnose as d then {
              // ICD-10 / ICD-11 aus Condition.code Slices
              cond.code as code then {
                code.coding as c where (system = 'http://fhir.de/CodeSystem/bfarm/icd-10-gm') then {
                  c.code as cd -> d.ICD_10 = cd "SetICD10";
                } "MapICD10";
                code.coding as c where (system = 'http://id.who.int/icd/release/11/mms') then {
                  c.code as cd -> d.ICD_11 = cd "SetICD11";
                } "MapICD11";
              } "MapCondCode";
              // Seitenlokalisation aus bodySite
              cond.bodySite as bs then {
                bs.coding as c then {
                  c.code as cd where ($this = '7771000') -> d.Seitenlokalisation = 'L' "SetSeiteL";
                  c.code as cd where ($this = '24028007') -> d.Seitenlokalisation = 'R' "SetSeiteR";
                  c.code as cd where ($this = '51440002') -> d.Seitenlokalisation = 'B' "SetSeiteB";
                } "MapSeiteCoding";
              } "MapSeite";
              // Diagnosedatum aus extension[Feststellungsdatum]
              cond.extension as ext where (url = 'https://www.medizininformatik-initiative.de/fhir/ext/modul-onko/StructureDefinition/mii-ex-onko-feststellungsdatum') then {
                ext.value as v -> d.Diagnosedatum = v "SetDiagnosedatum";
              } "MapDiagnoseDatum";
              // Diagnosesicherheit: TODO — VerificationStatus hat mehrere Codings; muss
              // gezielt nach oBDS-Mapping-System gefiltert werden (z.B. mii-vs-onko-primaertumor-diagnosesicherung).
              // -- cTNM aus klinischer TNM-Observation (LOINC 21908-9) --
              src.entry as ctnmEntry where (resource.is(Observation) and resource.code.coding.exists(code = '21908-9')) then {
                ctnmEntry.resource as ctnm then {
                  ctnm.component as comp where (code.coding.exists(code = '21905-5')) then {
                    comp.value as val then {
                      val.coding as c where (system = 'https://www.uicc.org/resources/tnm') then {
                        c.code as cd -> d.Praeth_T = cd "SetPraethT";
                      } "MapPraethTCode";
                    } "MapPraethTVal";
                  } "PraethT";
                  ctnm.component as comp where (code.coding.exists(code = '21906-3')) then {
                    comp.value as val then {
                      val.coding as c where (system = 'https://www.uicc.org/resources/tnm') then {
                        c.code as cd -> d.Praeth_N = cd "SetPraethN";
                      } "MapPraethNCode";
                    } "MapPraethNVal";
                  } "PraethN";
                  ctnm.component as comp where (code.coding.exists(code = '21907-1')) then {
                    comp.value as val then {
                      val.coding as c where (system = 'https://www.uicc.org/resources/tnm') then {
                        c.code as cd -> d.Praeth_M = cd "SetPraethM";
                      } "MapPraethMCode";
                    } "MapPraethMVal";
                  } "PraethM";
                } "MapCTNM";
              } "EntryCTNM";
              // -- pTNM aus pathologischer TNM-Observation (LOINC 21902-2) --
              src.entry as ptnmEntry where (resource.is(Observation) and resource.code.coding.exists(code = '21902-2')) then {
                ptnmEntry.resource as ptnm then {
                  ptnm.component as comp where (code.coding.exists(code = '21899-0')) then {
                    comp.value as val then {
                      val.coding as c where (system = 'https://www.uicc.org/resources/tnm') then {
                        c.code as cd -> d.Postop_T = cd "SetPostopT";
                      } "MapPostopTCode";
                    } "MapPostopTVal";
                  } "PostopT";
                  ptnm.component as comp where (code.coding.exists(code = '21900-6')) then {
                    comp.value as val then {
                      val.coding as c where (system = 'https://www.uicc.org/resources/tnm') then {
                        c.code as cd -> d.Postop_N = cd "SetPostopN";
                      } "MapPostopNCode";
                    } "MapPostopNVal";
                  } "PostopN";
                  ptnm.component as comp where (code.coding.exists(code = '21901-4')) then {
                    comp.value as val then {
                      val.coding as c where (system = 'https://www.uicc.org/resources/tnm') then {
                        c.code as cd -> d.Postop_M = cd "SetPostopM";
                      } "MapPostopMCode";
                    } "MapPostopMVal";
                  } "PostopM";
                } "MapPTNM";
              } "EntryPTNM";
            } "MapDiagnose";
            // -- Histologie: pro Pathologie-Befund ein Histologie-Block --
            src.entry as histEntry where (resource.is(Observation) and resource.code.coding.exists(code = '59847-4')) then {
              histEntry.resource as histObs -> f.Histologie as h then {
                histObs.effective as eff where ($this.is(dateTime)) -> h.Tumor_Histologiedatum = eff "SetHistDatum";
                histObs.value as val then {
                  val.coding as c where (system = 'urn:oid:2.16.840.1.113883.6.43.1') then {
                    c.code as cd -> h.Tumortyp = cd "SetTumortyp";
                  } "MapTumortypCoding";
                } "MapTumortyp";
              } "MapHistEntry";
            } "EntryHist";
            // -- Biomarker (ER/PR/HER2) als separate Histologie-Eintraege --
            // OncoBox 2.0: Anteil_positiv_Oestrogen, IRS_Oestrogen,
            // HormonrezeptorStatus_Oestrogen sind SEPARATE Felder
            // ER-Status → Histologie.HormonrezeptorStatus_Oestrogen (P/N/U via translate)
            src.entry as erEntry where (resource.is(Observation) and resource.code.coding.exists(code = '40556-3')) then {
              erEntry.resource as erObs -> f.Histologie as h then {
                erObs.value as val then {
                  val.coding as c where (system = 'http://loinc.org') -> h.HormonrezeptorStatus_Oestrogen = translate(c, 'https://www.senologie.org/fhir/ConceptMap/cm-loinc-rezeptorstatus-to-obds', 'code') "TranslateER";
                } "MapERValue";
                erObs.component as comp where (code.coding.exists(code = '1234804006')) then {
                  comp.value as val then {
                    val.value as v -> h.Anteil_positiv_Oestrogen = v "SetERAnteil";
                  } "MapERAnteilVal";
                } "ERAnteil";
                erObs.component as comp where (code.coding.exists(code = 'irs-score')) then {
                  comp.value as val then {
                    val.value as v -> h.IRS_Oestrogen = v "SetERIRS";
                  } "MapERIRSVal";
                } "ERIRS";
              } "MapEREntry";
            } "EntryER";
            // PR-Status → Histologie.HormonrezeptorStatus_Progesteron (P/N/U via translate)
            src.entry as prEntry where (resource.is(Observation) and resource.code.coding.exists(code = '85339-0')) then {
              prEntry.resource as prObs -> f.Histologie as h then {
                prObs.value as val then {
                  val.coding as c where (system = 'http://loinc.org') -> h.HormonrezeptorStatus_Progesteron = translate(c, 'https://www.senologie.org/fhir/ConceptMap/cm-loinc-rezeptorstatus-to-obds', 'code') "TranslatePR";
                } "MapPRValue";
                prObs.component as comp where (code.coding.exists(code = '1234803000')) then {
                  comp.value as val then {
                    val.value as v -> h.Anteil_positiv_Progesteron = v "SetPRAnteil";
                  } "MapPRAnteilVal";
                } "PRAnteil";
                prObs.component as comp where (code.coding.exists(code = 'irs-score')) then {
                  comp.value as val then {
                    val.value as v -> h.IRS_Progesteron = v "SetPRIRS";
                  } "MapPRIRSVal";
                } "PRIRS";
              } "MapPREntry";
            } "EntryPR";
            // HER2-Status → Histologie.Her2neuStatus
            // OncoBox 2.0 kennt keine P/N-Werte, sondern die IHC-Stufe (IHC0, IHC0-1, IHC1, IHC2-, IHC2+, IHC3);
            // HER2-low bleibt damit erhalten. Quelle ist der IHC-Score (LOINC 85319-2), bei 2+ zusaetzlich
            // Der IHC-Score kommt als LOINC-Antwort (Profil) oder als MII-MTB-Code 0-3 (Pathologie-Formular),
            // ultralow als MII-Leitliniencode oder lokaler Formularcode.
            // das ISH-Ergebnis (LOINC 96893-3, im Formular 85318-4; Werte als LOINC-Antwort oder SNOMED CT). 2+ ohne ISH-Ergebnis → 'U'.
            src.entry as her2Entry where (resource.is(Observation) and resource.code.coding.exists(code = '48676-1')) then {
              her2Entry.resource as her2Obs -> f.Histologie as h then {
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA6111-4') or (system.endsWith('her2-ihc-score') and (code = '0'))) and value.coding.exists((code = 'ultralow') or (code = 'her2-ultralow')).not()) -> h.Her2neuStatus = 'IHC0' "SetHER2IHC0";
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA6111-4') or (system.endsWith('her2-ihc-score') and (code = '0'))) and value.coding.exists((code = 'ultralow') or (code = 'her2-ultralow'))) -> h.Her2neuStatus = 'IHC0-1' "SetHER2IHC0plus";
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA11841-6') or (system.endsWith('her2-ihc-score') and (code = '1')))) -> h.Her2neuStatus = 'IHC1' "SetHER2IHC1";
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA11842-4') or (system.endsWith('her2-ihc-score') and (code = '2'))) and component.where(code.coding.exists((code = '96893-3') or (code = '85318-4'))).value.coding.exists((code = 'LA6577-6') or (code = '260385009'))) -> h.Her2neuStatus = 'IHC2-' "SetHER2IHC2neg";
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA11842-4') or (system.endsWith('her2-ihc-score') and (code = '2'))) and component.where(code.coding.exists((code = '96893-3') or (code = '85318-4'))).value.coding.exists((code = 'LA6576-8') or (code = '10828004'))) -> h.Her2neuStatus = 'IHC2+' "SetHER2IHC2pos";
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA11842-4') or (system.endsWith('her2-ihc-score') and (code = '2'))) and component.where(code.coding.exists((code = '96893-3') or (code = '85318-4'))).value.coding.exists((code = 'LA6577-6') or (code = 'LA6576-8') or (code = '260385009') or (code = '10828004')).not()) -> h.Her2neuStatus = 'U' "SetHER2IHC2ohneISH";
                her2Obs where (component.where(code.coding.exists(code = '85319-2')).value.coding.exists((code = 'LA11843-2') or (system.endsWith('her2-ihc-score') and (code = '3')))) -> h.Her2neuStatus = 'IHC3' "SetHER2IHC3";
              } "MapHER2Entry";
            } "EntryHER2";
            // -- Ki-67 (LOINC 85318-4) --
            src.entry as ki67Entry where (resource.is(Observation) and resource.code.coding.exists(code = '85318-4')) then {
              ki67Entry.resource as ki67Obs -> f.Histologie as h then {
                ki67Obs.value as val then {
                  val.value as v -> h.Ki_67 = v "SetKi67";
                } "MapKi67Val";
              } "MapKi67Entry";
            } "EntryKi67";
            // -- Grading (LOINC 33732-9) --
            src.entry as gradEntry where (resource.is(Observation) and resource.code.coding.exists(code = '33732-9')) then {
              gradEntry.resource as gradObs -> f.Histologie as h then {
                gradObs.value as val then {
                  val.coding as c then {
                    c.code as cd -> h.Grading = cd "SetGrading";
                  } "MapGradingCoding";
                } "MapGradingVal";
              } "MapGradingEntry";
            } "EntryGrad";
            // -- Lymphknoten untersucht/befallen, Sentinel-LK --
            src.entry as lkuEntry where (resource.is(Observation) and resource.code.coding.exists(code = '21894-1')) then {
              lkuEntry.resource as lkuObs -> f.Histologie as h then {
                lkuObs.value as val then {
                  val.value as v -> h.LK_untersucht = truncate(v) "SetLKUnt";
                } "MapLKUntVal";
              } "MapLKUntEntry";
            } "EntryLKUnt";
            src.entry as lkbEntry where (resource.is(Observation) and resource.code.coding.exists(code = '21893-3')) then {
              lkbEntry.resource as lkbObs -> f.Histologie as h then {
                lkbObs.value as val then {
                  val.value as v -> h.LK_befallen = truncate(v) "SetLKBef";
                } "MapLKBefVal";
              } "MapLKBefEntry";
            } "EntryLKBef";
            src.entry as slkuEntry where (resource.is(Observation) and resource.code.coding.exists(code = '92832-5')) then {
              slkuEntry.resource as slkuObs -> f.Histologie as h then {
                slkuObs.value as val then {
                  val.value as v -> h.Sentinel_LK_untersucht = truncate(v) "SetSLKUnt";
                } "MapSLKUntVal";
              } "MapSLKUntEntry";
            } "EntrySLKUnt";
            src.entry as slkbEntry where (resource.is(Observation) and resource.code.coding.exists(code = '92833-3')) then {
              slkbEntry.resource as slkbObs -> f.Histologie as h then {
                slkbObs.value as val then {
                  val.value as v -> h.Sentinel_LK_befallen = truncate(v) "SetSLKBef";
                } "MapSLKBefVal";
              } "MapSLKBefEntry";
